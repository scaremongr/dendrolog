cmake_minimum_required(VERSION 3.16)
project(ViewExportTest CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(viewexport_test main.cpp ../../src/viewexport.cpp)
target_include_directories(viewexport_test PRIVATE ../../src)
target_link_libraries(viewexport_test PRIVATE Qt6::Core)
add_test(NAME viewexport COMMAND viewexport_test)
set_tests_properties(viewexport PROPERTIES TIMEOUT 30)
if(WIN32)
    get_target_property(_qt_core_loc Qt6::Core LOCATION)
    get_filename_component(_qt_bin_dir "${_qt_core_loc}" DIRECTORY)
    set_tests_properties(viewexport PROPERTIES
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qt_bin_dir}")
endif()
